// =====================================================
// FRAGMENTO COCHABAMBA
// SQL Server
// =====================================================

Project Hospital_CBBA {
  database_type: "SQLServer"
}

// =====================================================
// PACIENTES CBBA
// Fragmentación Horizontal
// =====================================================

Table paciente_cbba {
  id_paciente integer [pk]
  nombre varchar
  apellido varchar
  fecha_nacimiento date
  tipo_sangre varchar
  alergias text
  enfermedades_cronicas text
  telefono varchar
  direccion text

  id_hospital integer

  Note: '''
  Fragmento horizontal:
  SOLO pacientes de Cochabamba.
  '''
}

// =====================================================
// HISTORIAL LOCAL
// =====================================================

Table historial_cbba {
  id_historial integer [pk]
  id_paciente integer
  observaciones text
  fecha datetime
}

// =====================================================
// CONSULTAS LOCALES
// =====================================================

Table consulta_cbba {
  id_consulta integer [pk]
  id_paciente integer
  diagnostico text
  tratamiento text
  fecha datetime
}

// =====================================================
// REPLICA CRITICA
// =====================================================

Table replica_critica_cbba {
  id_replica integer [pk]

  id_paciente integer
  hospital_origen varchar

  tipo_sangre varchar
  alergias text
  enfermedades_cronicas text

  fecha_actualizacion datetime
}

// =====================================================
// RELACIONES
// =====================================================

Ref: historial_cbba.id_paciente > paciente_cbba.id_paciente

Ref: consulta_cbba.id_paciente > paciente_cbba.id_paciente
