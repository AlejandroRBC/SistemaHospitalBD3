// =====================================================
// FRAGMENTO SANTA CRUZ
// SQL Server
// =====================================================

Project Hospital_STCZ {
  database_type: "SQLServer"
}

// =====================================================
// PACIENTES STCZ
// Fragmentación Horizontal
// =====================================================

Table paciente_stcz {
  id_paciente integer [pk]
  nombre varchar
  apellido varchar
  fecha_nacimiento date
  tipo_sangre varchar
  alergias text
  enfermedades_cronicas text
  telefono varchar
  direccion text

  id_hospital integer

  Note: '''
  Fragmento horizontal:
  SOLO pacientes de Santa Cruz.
  '''
}

// =====================================================
// HISTORIAL LOCAL
// =====================================================

Table historial_stcz {
  id_historial integer [pk]
  id_paciente integer
  observaciones text
  fecha datetime
}

// =====================================================
// CONSULTAS LOCALES
// =====================================================

Table consulta_stcz {
  id_consulta integer [pk]
  id_paciente integer
  diagnostico text
  tratamiento text
  fecha datetime
}

// =====================================================
// REPLICA CRITICA
// =====================================================

Table replica_critica_stcz {
  id_replica integer [pk]

  id_paciente integer
  hospital_origen varchar

  tipo_sangre varchar
  alergias text
  enfermedades_cronicas text

  fecha_actualizacion datetime
}

// =====================================================
// RELACIONES
// =====================================================

Ref: historial_stcz.id_paciente > paciente_stcz.id_paciente

Ref: consulta_stcz.id_paciente > paciente_stcz.id_paciente
